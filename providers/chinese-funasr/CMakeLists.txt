cmake_minimum_required(VERSION 3.20)
project(autobyteus_funasr_provider LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "Current Fun-ASR production package requires Apple/CommonCrypto")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
if(MSVC)
  add_compile_options("/pathmap:${LLAMA_CPP_SOURCE_DIR}=llama.cpp" "/pathmap:${UTF8PROC_SOURCE_DIR}=utf8proc" "/pathmap:${CMAKE_SOURCE_DIR}=provider" "/pathmap:${CMAKE_BINARY_DIR}=build")
else()
  add_compile_options("-ffile-prefix-map=${LLAMA_CPP_SOURCE_DIR}=llama.cpp" "-ffile-prefix-map=${UTF8PROC_SOURCE_DIR}=utf8proc" "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=provider" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_CPP_SOURCE_DIR} llama-cpp)
add_executable(voice-provider-worker src/main.cpp src/session.cpp src/package_integrity.cpp src/package_integrity_apple.cpp src/preparation_diagnostics.cpp src/audio.cpp src/audio_features.cpp src/normalization.cpp src/result_policy.cpp src/funasr_engine.cpp ${UTF8PROC_SOURCE_DIR}/utf8proc.c)
target_include_directories(voice-provider-worker PRIVATE src ${LLAMA_CPP_SOURCE_DIR}/vendor ${LLAMA_CPP_SOURCE_DIR}/common ${UTF8PROC_SOURCE_DIR})
target_link_libraries(voice-provider-worker PRIVATE llama ggml)
if(MSVC)
  target_compile_options(voice-provider-worker PRIVATE /W4 /WX)
  target_link_options(voice-provider-worker PRIVATE /Brepro)
else()
  target_compile_options(voice-provider-worker PRIVATE -Wall -Wextra -Werror)
endif()
option(VOICE_PROVIDER_BUILD_TESTS "Build provider contract tests" OFF)
if(VOICE_PROVIDER_BUILD_TESTS)
  enable_testing()
  add_executable(voice-provider-contract-test tests/normalization_and_result_policy_test.cpp src/normalization.cpp src/result_policy.cpp ${UTF8PROC_SOURCE_DIR}/utf8proc.c)
  target_include_directories(voice-provider-contract-test PRIVATE src ${LLAMA_CPP_SOURCE_DIR}/vendor ${UTF8PROC_SOURCE_DIR})
  if(MSVC)
    target_compile_options(voice-provider-contract-test PRIVATE /W4 /WX)
  else()
    target_compile_options(voice-provider-contract-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME voice-provider-contract-test COMMAND voice-provider-contract-test ${CMAKE_SOURCE_DIR}/../../contracts/normalization/twp-to-cn-v1.json ${CMAKE_SOURCE_DIR}/../../contracts/normalization/fixtures-v1.json)
endif()
